`timescale 1ns/1ps

module test;

reg [3:0] A;
reg [3:0] B;
reg Cin;

wire [3:0] Sum;
wire Cout;

ripple_carry_adder uut (
    .A(A),
    .B(B),
    .Cin(Cin),
    .Sum(Sum),
    .Cout(Cout)
);

initial begin

    // Waveform generation
    $dumpfile("dump.vcd");
    $dumpvars(0, test);

    $display("Time\tA\tB\tCin\tSum\tCout");

    A = 4'b0000; B = 4'b0000; Cin = 0;
    #10;
    $display("%0t\t%b\t%b\t%b\t%b\t%b",$time,A,B,Cin,Sum,Cout);

    A = 4'b0011; B = 4'b0101; Cin = 0;
    #10;
    $display("%0t\t%b\t%b\t%b\t%b\t%b",$time,A,B,Cin,Sum,Cout);

    A = 4'b1111; B = 4'b0001; Cin = 0;
    #10;
    $display("%0t\t%b\t%b\t%b\t%b\t%b",$time,A,B,Cin,Sum,Cout);

    A = 4'b1010; B = 4'b0110; Cin = 1;
    #10;
    $display("%0t\t%b\t%b\t%b\t%b\t%b",$time,A,B,Cin,Sum,Cout);

    A = 4'b1111; B = 4'b1111; Cin = 0;
    #10;
    $display("%0t\t%b\t%b\t%b\t%b\t%b",$time,A,B,Cin,Sum,Cout);

    #10;
    $finish;

end

endmodule
